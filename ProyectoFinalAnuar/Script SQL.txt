CREATE DATABASE ClinicaDB;
GO

USE ClinicaDB;
GO

-- =========================
-- 1. ROLES Y USUARIOS
-- =========================

CREATE TABLE Roles (
    RoleId      INT IDENTITY(1,1) PRIMARY KEY,
    Name        VARCHAR(50) NOT NULL,
    Description VARCHAR(200) NULL
);

INSERT INTO Roles (Name, Description) VALUES
('RecursosHumanos', 'Gestión de empleados y accesos'),
('Administrativo', 'Gestión de pacientes y facturación'),
('SoporteInformacion', 'Gestión inventarios e información'),
('Enfermera', 'Registro de signos vitales y aplicaciones'),
('Medico', 'Historia clínica y órdenes médicas');

CREATE TABLE Users (
    UserId          INT IDENTITY(1,1) PRIMARY KEY,
    Cedula          VARCHAR(10) NOT NULL UNIQUE,
    NombreCompleto  VARCHAR(100) NOT NULL,
    Email           VARCHAR(100) NOT NULL,
    Telefono        VARCHAR(10) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    Direccion       VARCHAR(30) NOT NULL,
    RoleId          INT NOT NULL,
    Username        VARCHAR(15) NOT NULL UNIQUE,
    PasswordHash    VARCHAR(200) NOT NULL,
    CONSTRAINT FK_Users_Roles FOREIGN KEY(RoleId) REFERENCES Roles(RoleId),
    CONSTRAINT CK_Users_Telefono_Numerico CHECK (Telefono NOT LIKE '%[^0-9]%')
);
GO

-- =========================
-- 2. PACIENTES Y ADMINISTRATIVO
-- =========================

CREATE TABLE Patients (
    PatientId       INT IDENTITY(1,1) PRIMARY KEY,
    Cedula          VARCHAR(10) NOT NULL UNIQUE,
    NombreCompleto  VARCHAR(100) NOT NULL,
    FechaNacimiento DATE NOT NULL,
    Genero          VARCHAR(10) NOT NULL, -- 'Masculino','Femenino','Otro'
    Direccion       VARCHAR(200) NOT NULL,
    Telefono        VARCHAR(10) NOT NULL,
    Email           VARCHAR(100) NULL,
    CONSTRAINT CK_Patients_Telefono_Numerico CHECK (Telefono NOT LIKE '%[^0-9]%')
);
GO

CREATE TABLE EmergencyContact (
    EmergencyContactId INT IDENTITY(1,1) PRIMARY KEY,
    PatientId          INT NOT NULL,
    Nombres            VARCHAR(50) NOT NULL,
    Apellidos          VARCHAR(50) NOT NULL,
    Relacion           VARCHAR(50) NOT NULL,
    Telefono           VARCHAR(10) NOT NULL,
    CONSTRAINT FK_EmergencyContact_Patients FOREIGN KEY(PatientId) REFERENCES Patients(PatientId),
    CONSTRAINT CK_EmergencyContact_Tel_Numerico CHECK (Telefono NOT LIKE '%[^0-9]%'),
    CONSTRAINT UQ_EmergencyContact_Patient UNIQUE (PatientId)
);
GO

CREATE TABLE InsurancePolicy (
    PolicyId    INT IDENTITY(1,1) PRIMARY KEY,
    PatientId   INT NOT NULL UNIQUE, -- una póliza por paciente
    CompanyName VARCHAR(100) NOT NULL,
    PolicyNumber VARCHAR(50) NOT NULL,
    IsActive    BIT NOT NULL,
    EndDate     DATE NOT NULL,
    CONSTRAINT FK_InsurancePolicy_Patients FOREIGN KEY(PatientId) REFERENCES Patients(PatientId)
);
GO

-- =========================
-- 3. INVENTARIOS E ESPECIALIDADES
-- =========================

CREATE TABLE SpecialtyCatalog (
    SpecialtyId INT IDENTITY(1,1) PRIMARY KEY,
    Nombre      VARCHAR(100) NOT NULL
);

CREATE TABLE MedicationCatalog (
    MedicationId INT IDENTITY(1,1) PRIMARY KEY,
    Nombre       VARCHAR(100) NOT NULL,
    CostoBase    DECIMAL(12,2) NOT NULL,
    Description  VARCHAR(500) NULL,
    IsActive     BIT NOT NULL DEFAULT(1)
);

CREATE TABLE ProcedureCatalog (
    ProcedureId INT IDENTITY(1,1) PRIMARY KEY,
    Nombre      VARCHAR(100) NOT NULL,
    CostoBase   DECIMAL(12,2) NOT NULL
);

CREATE TABLE DiagnosticHelpCatalog (
    DiagnosticHelpId INT IDENTITY(1,1) PRIMARY KEY,
    Nombre           VARCHAR(100) NOT NULL,
    CostoBase        DECIMAL(12,2) NOT NULL
);
GO

-- =========================
-- 4. ÓRDENES MÉDICAS
-- =========================

CREATE TABLE Orders (
    OrderNumber    INT PRIMARY KEY,         -- máximo 6 dígitos: lo valida la app
    CedulaPaciente VARCHAR(10) NOT NULL,    -- FK lógica a Patients.Cedula
    CedulaMedico   VARCHAR(10) NOT NULL,    -- FK lógica a Users.Cedula (rol Medico)
    CreatedAt      DATETIME NOT NULL DEFAULT(GETDATE())
);
GO

-- Medicamentos
CREATE TABLE OrderMedication (
    OrderNumber   INT NOT NULL,
    ItemNumber    INT NOT NULL,
    MedicationId  INT NOT NULL,
    Dosis         VARCHAR(100) NOT NULL,
    Duracion      VARCHAR(100) NOT NULL,
    Costo         DECIMAL(12,2) NOT NULL,
    CONSTRAINT PK_OrderMedication PRIMARY KEY (OrderNumber, ItemNumber),
    CONSTRAINT FK_OrderMedication_Orders FOREIGN KEY(OrderNumber) REFERENCES Orders(OrderNumber),
    CONSTRAINT FK_OrderMedication_MedCatalog FOREIGN KEY(MedicationId) REFERENCES MedicationCatalog(MedicationId)
);
GO

-- Procedimientos
CREATE TABLE OrderProcedure (
    OrderNumber       INT NOT NULL,
    ItemNumber        INT NOT NULL,
    ProcedureId       INT NOT NULL,
    VecesRepite       INT NOT NULL,
    Frecuencia        VARCHAR(50) NOT NULL,
    RequiresSpecialist BIT NOT NULL,
    SpecialtyId       INT NULL,
    Costo             DECIMAL(12,2) NOT NULL,
    CONSTRAINT PK_OrderProcedure PRIMARY KEY (OrderNumber, ItemNumber),
    CONSTRAINT FK_OrderProcedure_Orders FOREIGN KEY(OrderNumber) REFERENCES Orders(OrderNumber),
    CONSTRAINT FK_OrderProcedure_ProcedureCatalog FOREIGN KEY(ProcedureId) REFERENCES ProcedureCatalog(ProcedureId),
    CONSTRAINT FK_OrderProcedure_SpecialtyCatalog FOREIGN KEY(SpecialtyId) REFERENCES SpecialtyCatalog(SpecialtyId)
);
GO

-- Ayudas diagnósticas
CREATE TABLE OrderDiagnosticHelp (
    OrderNumber       INT NOT NULL,
    ItemNumber        INT NOT NULL,
    DiagnosticHelpId  INT NOT NULL,
    Cantidad          INT NOT NULL,
    RequiresSpecialist BIT NOT NULL,
    SpecialtyId       INT NULL,
    Costo             DECIMAL(12,2) NOT NULL,
    CONSTRAINT PK_OrderDiagnosticHelp PRIMARY KEY (OrderNumber, ItemNumber),
    CONSTRAINT FK_OrderDiagnosticHelp_Orders FOREIGN KEY(OrderNumber) REFERENCES Orders(OrderNumber),
    CONSTRAINT FK_OrderDiagnosticHelp_DiagCatalog FOREIGN KEY(DiagnosticHelpId) REFERENCES DiagnosticHelpCatalog(DiagnosticHelpId),
    CONSTRAINT FK_OrderDiagnosticHelp_SpecialtyCatalog FOREIGN KEY(SpecialtyId) REFERENCES SpecialtyCatalog(SpecialtyId)
);
GO

-- =========================
-- 5. VISITAS DE ENFERMERÍA
-- =========================

CREATE TABLE NurseVisit (
    VisitId        INT IDENTITY(1,1) PRIMARY KEY,
    CedulaPaciente VARCHAR(10) NOT NULL,
    CedulaEnfermera VARCHAR(10) NOT NULL,
    FechaHora      DATETIME NOT NULL DEFAULT(GETDATE()),
    PresionArterial VARCHAR(20) NOT NULL,
    Temperatura     DECIMAL(4,1) NOT NULL,
    Pulso           INT NOT NULL,
    Oxigeno         INT NOT NULL,
    Observaciones   VARCHAR(MAX) NULL,
    OrderNumber     INT NULL,
    ItemNumber      INT NULL,
    CONSTRAINT FK_NurseVisit_Orders FOREIGN KEY(OrderNumber) REFERENCES Orders(OrderNumber)
);
GO

-- =========================
-- 6. FACTURACIÓN
-- =========================

CREATE TABLE InvoiceHeader (
    InvoiceId         INT IDENTITY(1,1) PRIMARY KEY,
    PatientId         INT NOT NULL,
    CedulaMedico      VARCHAR(10) NOT NULL,
    PolicyId          INT NULL,
    CompanyName       VARCHAR(100) NULL,
    PolicyNumber      VARCHAR(50) NULL,
    PolicyEndDate     DATE NULL,
    PolicyDaysRemaining INT NULL,
    InvoiceDate       DATETIME NOT NULL DEFAULT(GETDATE()),
    TotalAmount       DECIMAL(12,2) NOT NULL,
    CopayAmount       DECIMAL(12,2) NOT NULL,
    InsurerAmount     DECIMAL(12,2) NOT NULL,
    [Year]            INT NOT NULL,
    CONSTRAINT FK_InvoiceHeader_Patient FOREIGN KEY(PatientId) REFERENCES Patients(PatientId),
    CONSTRAINT FK_InvoiceHeader_Policy FOREIGN KEY(PolicyId) REFERENCES InsurancePolicy(PolicyId)
);
GO

CREATE TABLE InvoiceDetail (
    InvoiceDetailId INT IDENTITY(1,1) PRIMARY KEY,
    InvoiceId       INT NOT NULL,
    Description     VARCHAR(200) NOT NULL,
    Quantity        INT NOT NULL,
    UnitCost        DECIMAL(12,2) NOT NULL,
    LineTotal       DECIMAL(12,2) NOT NULL,
    ItemType        VARCHAR(20) NOT NULL, -- 'Medicamento','Procedimiento','Ayuda'
    CONSTRAINT FK_InvoiceDetail_InvoiceHeader FOREIGN KEY(InvoiceId) REFERENCES InvoiceHeader(InvoiceId)
);
GO
